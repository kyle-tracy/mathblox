> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the free-walk Road (`?craft`) with a path map (`?map`): a graph of nodes and paths spiralling out from Sumhenge, travelled one node per arrow press, with two bonus rooms.

**Architecture:** A pure graph in `core/map.js` (nodes, two-way links, `step` → events) reusing `core/progress.js` for unlocks. `render/map.js` draws it and owns its input (as `craft/play.js` did). Bonus rooms are ordinary levels in ordinary `WORLDS` entries. The old Road code is deleted; git keeps it.

**Tech Stack:** Plain ES modules, canvas, `node --test`. No dependencies.

**Spec:** `docs/superpowers/specs/2026-10-27-path-map-design.md` (meta-theme: graph theory).

// The path map (spec 2026-10-27-path-map-design.md): a graph you travel, never a play area.
import { test } from 'node:test';
import assert from 'node:assert/strict';
import { NODES, newMap, step } from '../core/map.js';
import { WORLDS } from '../levels/worlds.js';

const OPP = { u: 'd', d: 'u', l: 'r', r: 'l' };
const nodeOf = id => Object.keys(NODES).find(k => NODES[k].world === id);

test('every link is two-way, in the opposite direction', () => {
  for (const [a, n] of Object.entries(NODES))
    for (const [d, b] of Object.entries(n.links)) assert.equal(NODES[b]?.links[OPP[d]], a, `${a} -${d}-> ${b} has no way back`);
});

test('the graph is connected from Sumhenge, and every world sits on exactly one node', () => {
  const seen = new Set(['sumhenge']), todo = ['sumhenge'];
  while (todo.length) for (const b of Object.values(NODES[todo.pop()].links)) if (!seen.has(b)) { seen.add(b); todo.push(b); }
  assert.deepEqual([...seen].sort(), Object.keys(NODES).sort());
  for (const w of WORLDS) assert.equal(Object.values(NODES).filter(n => n.world === w.id).length, 1, w.id);
});

test('walking never fails: a missing link is a no-op, a link moves you', () => {
  const s = newMap();
  assert.deepEqual(step(s, 'l').events, []);
  const r = step(s, 'r');
  assert.equal(r.state.at, NODES.sumhenge.links.r);
  assert.deepEqual(r.events, [{ type: 'walk', from: 'sumhenge', to: NODES.sumhenge.links.r }]);
});

test('the demo order: each world enters once the ones before it are done', () => {
  const done = [];
  for (const id of ['camp', 'alexandria', 'croton', 'ujjain', 'elea', 'keep-euclid', 'keep-pythagoras', 'keep-brahmagupta', 'keep-zeno', 'dungeon']) {
    assert.equal(step(newMap(done, nodeOf(id)), 'enter').events[0].type, 'enter', `${id} after ${done}`);
    done.push(id);
  }
});

test('a locked world says what it needs; an empty world is sealed beyond the demo', () => {
  assert.deepEqual(step(newMap([], nodeOf('croton')), 'enter').events, [{ type: 'locked', world: 'croton', missing: ['euclid'] }]);
  assert.deepEqual(step(newMap([], nodeOf('rotation')), 'enter').events, [{ type: 'locked', world: 'rotation', missing: [], soon: true }]);
  assert.deepEqual(step(newMap(), 'keep').events ?? [], []); // junctions and bad input do nothing

```js
// The path map (docs/superpowers/specs/2026-10-27-path-map-design.md): Sumhenge is the root, worlds are vertices,
// paths are edges. Travel only: one arrow walks one edge, Enter opens a node's world. Nothing to push, nothing to strand.
// Coordinates are layout units for render/map.js (Arto may move them; the links are the map). Pure, no DOM.
import { world, missing, recruits } from './progress.js';

export const NODES = {
  sumhenge: { x: 0, y: 0, world: 'camp', links: { r: 'alexandria', d: 'campfire', u: 'under' } },
  campfire: { x: 0, y: 1, world: 'campfire', links: { u: 'sumhenge' } },
  under: { x: 0, y: -1, world: 'under', links: { d: 'sumhenge' } },
  alexandria: { x: 2, y: 0, world: 'alexandria', links: { l: 'sumhenge', d: 'croton', r: 'vectors' } },
  vectors: { x: 4, y: 0, world: 'vectors', links: { l: 'alexandria' } },
  croton: { x: 2, y: 2, world: 'croton', links: { u: 'alexandria', l: 'hut', r: 'rotation' } },
  rotation: { x: 4, y: 2, world: 'rotation', links: { l: 'croton' } },
  hut: { x: 0, y: 2, world: 'hut', links: { r: 'croton', l: 'ujjain' } },
  ujjain: { x: -2, y: 2, world: 'ujjain', links: { r: 'hut', u: 'elea', d: 'distrib' } },
  distrib: { x: -2, y: 4, world: 'distrib', links: { u: 'ujjain' } },
  elea: { x: -2, y: -2, world: 'elea', links: { d: 'ujjain', r: 'keep', l: 'series', u: 'motion' } },
  series: { x: -4, y: -2, world: 'series', links: { r: 'elea' } },
  motion: { x: -2, y: -4, world: 'motion', links: { d: 'elea' } },
  keep: { x: 6, y: -2, links: { l: 'elea', d: 'stair1' } }, // the Storm Keep's chhatri; the stepwell goes down
  stair1: { x: 6, y: -1, links: { u: 'keep', l: 'keep-euclid-n', r: 'keep-pythagoras-n', d: 'stair2' } },
  'keep-euclid-n': { x: 5, y: -1, world: 'keep-euclid', links: { r: 'stair1' } },
  'keep-pythagoras-n': { x: 7, y: -1, world: 'keep-pythagoras', links: { l: 'stair1' } },
  stair2: { x: 6, y: 0, links: { u: 'stair1', l: 'keep-brahmagupta-n', r: 'keep-zeno-n', d: 'pool' } },
  'keep-brahmagupta-n': { x: 5, y: 0, world: 'keep-brahmagupta', links: { r: 'stair2' } },
  'keep-zeno-n': { x: 7, y: 0, world: 'keep-zeno', links: { l: 'stair2' } },
  pool: { x: 6, y: 1, world: 'dungeon', links: { u: 'stair2' } },
};

export const newMap = (done = [], at = 'sumhenge') => ({ at, done: [...done] });

// dir: 'u' | 'd' | 'l' | 'r' walks an edge; 'enter' opens the node's world. Anything else does nothing.
export function step(state, dir) {
  const node = NODES[state.at];
  if (dir === 'enter') {
    if (!node.world) return { state, events: [] };
    if (!world(node.world).levels.length) return { state, events: [{ type: 'locked', world: node.world, missing: [], soon: true }] };
    const miss = missing(node.world, recruits(state.done), state.done);
    return { state, events: [miss.length ? { type: 'locked', world: node.world, missing: miss } : { type: 'enter', world: node.world }] };
  }
  const to = node.links[dir];
  return to ? { state: { ...state, at: to }, events: [{ type: 'walk', from: state.at, to }] } : { state, events: [] };
}
```
